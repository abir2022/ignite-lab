# Virtual Deep Sea Drilling Controller
import drilling_lib as dsys
import time

def initialize_drill(depth_target):
    print("Calibrating system sensors...")
    dsys.set_depth(depth_target)
    return True

if __name__ == "__main__":
    target = 12000 # meters
    if initialize_drill(target):
        dsys.start_rotation(rpm=120)
        print("Operation started at 12,000m target.")
    else:
        print("Error: Hardware failure.")
